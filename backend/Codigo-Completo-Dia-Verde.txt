const HOJA_HISTORICO = 'Hoja 1';

function dvConfiguracion_(nombre) {
  const valor = PropertiesService.getScriptProperties().getProperty(nombre);
  if (!valor) throw new Error('Falta configurar la propiedad ' + nombre + ' del proyecto.');
  return valor;
}

function doPost(e) {
  const auth = dvProcesarAuth_(e);
  if (auth !== null) return auth;
  return hacerPost(e);
}

function hacerPost(e) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
    if (!e || !e.postData || !e.postData.contents) {
      throw new Error('No se recibieron datos desde la aplicación.');
    }
    const datos = JSON.parse(e.postData.contents);
    const fecha = normalizarFecha(String(datos.fecha || '').trim());
    const hora = String(datos.hora || '').trim();
    const cgm = String(datos.cgm || '').trim();
    const zona = String(datos.zona || '').trim();
    const direccion = String(datos.direccion || '').trim();
    const latitud = String(datos.latitud || '').trim();
    const longitud = String(datos.longitud || '').trim();
    const foto = String(datos.foto || '').trim();
    const idEvidencia = String(datos.idEvidencia || '').trim();
    if (!direccion) throw new Error('No se recibió la dirección.');

    const libro = SpreadsheetApp.openById(dvConfiguracion_('ID_HOJA_DE_DISPERSION'));
    const hoja = libro.getSheetByName(HOJA_HISTORICO);
    if (!hoja) throw new Error("No existe la hoja '" + HOJA_HISTORICO + "' en el archivo indicado.");
    if (hoja.getRange(1, 9).getValue() !== 'ID_EVIDENCIA') {
      hoja.getRange(1, 9).setValue('ID_EVIDENCIA');
    }
    const ultimaFila = hoja.getLastRow();
    if (idEvidencia && ultimaFila >= 2) {
      const ids = hoja.getRange(2, 9, ultimaFila - 1, 1).getDisplayValues();
      for (let j = 0; j < ids.length; j++) {
        if (String(ids[j][0] || '').trim() === idEvidencia) {
          return respuestaJSON({ok:true, duplicado:true, idEvidencia:idEvidencia,
            mensaje:'Esta evidencia ya fue procesada; no se duplicó el registro.', fila:j + 2});
        }
      }
    }
    if (ultimaFila >= 2) {
      const valores = hoja.getRange(2, 1, ultimaFila - 1, 8).getValues();
      for (let i = 0; i < valores.length; i++) {
        const fila = valores[i];
        if (normalizarFecha(fila[0]) === fecha && normalizarClave(fila[2]) === normalizarClave(zona) &&
            normalizarClave(fila[3]) === normalizarClave(direccion)) {
          return respuestaJSON({ok:true, yaRealizado:true, estado:'Realizado Hoy',
            fecha:normalizarFecha(fila[0]), hora:String(fila[1] || ''), cgm:cgm, zona:zona,
            direccion:direccion, latitud:String(fila[4] || latitud || ''),
            longitud:String(fila[5] || longitud || ''), foto:String(fila[6] || ''),
            mensaje:'La dirección ya fue realizada hoy.'});
        }
      }
    }

    const carpetaRaiz = DriveApp.getFolderById(dvConfiguracion_('CARPETA_DRIVE_ID'));
    const ahora = new Date();
    const zonaHoraria = Session.getScriptTimeZone();
    const anio = Utilities.formatDate(ahora, zonaHoraria, 'yyyy');
    const mes = Utilities.formatDate(ahora, zonaHoraria, 'MM - MMMM');
    const dia = Utilities.formatDate(ahora, zonaHoraria, 'dd-MM-yyyy');
    const carpetaAnio = obtenerOCrearCarpeta(carpetaRaiz, anio);
    const carpetaMes = obtenerOCrearCarpeta(carpetaAnio, mes);
    let carpetaDestino = obtenerOCrearCarpeta(carpetaMes, dia);
    if (cgm) carpetaDestino = obtenerOCrearCarpeta(carpetaDestino, cgm);
    if (zona) carpetaDestino = obtenerOCrearCarpeta(carpetaDestino, zona);
    let enlaceFoto = '';
    if (foto) {
      const partes = foto.split(',');
      const base64 = partes.length > 1 ? partes[1] : partes[0];
      let mime = 'image/jpeg';
      if (partes.length > 1 && partes[0].includes('data:')) {
        const coincidencia = partes[0].match(/data:(.*?);base64/);
        if (coincidencia) mime = coincidencia[1];
      }
      const fechaArchivo = fecha.replace(/[\/\\:]/g, '-').replace(/\s+/g, '_');
      const horaArchivo = hora.replace(/:/g, '-').replace(/\s+/g, '_');
      const blob = Utilities.newBlob(Utilities.base64Decode(base64), mime, fechaArchivo + '_' + horaArchivo + '.jpg');
      enlaceFoto = carpetaDestino.createFile(blob).getUrl();
    }
    // A:I = FECHA, HORA, ZONA, DIRECCION, LATITUD, LONGITUD, FOTO, ESTADO, ID_EVIDENCIA.
    hoja.appendRow([fecha, hora, zona, direccion, latitud, longitud, enlaceFoto, 'Realizado Hoy', idEvidencia]);
    SpreadsheetApp.flush();
    return respuestaJSON({ok:true, mensaje:'Evidencia guardada correctamente.', fecha:fecha,
      hora:hora, cgm:cgm, zona:zona, direccion:direccion, latitud:latitud, longitud:longitud,
      foto:enlaceFoto, estado:'Realizado Hoy', idEvidencia:idEvidencia});
  } catch (error) {
    console.error(error.stack || error.message);
    return respuestaJSON({ok:false, error:error.message});
  } finally {
    try { lock.releaseLock(); } catch (error) {}
  }
}

function doGet(e) { return obtener(e); }

function obtener(e) {
  try {
    const accion = e && e.parameter ? e.parameter.accion || '' : '';
    if (accion === 'registrarGPSLote') return registrarGPSLote(e);
    if (accion === 'registrarGPS') return registrarGPS(e);
    if (accion === 'evidenciasHoy' || accion === 'obtener') return obtenerEvidenciasHoy();
    return respuestaJSON({ok:true, mensaje:'DIA VERDE Apps Script funcionando'});
  } catch (error) { return respuestaJSON({ok:false, error:error.message}); }
}

function registrarGPSLote(e) {
  const lock = LockService.getScriptLock();
  try {
    const p = e && e.parameter ? e.parameter : {};
    if (!p.lote) throw new Error('No se recibió el lote GPS.');
    const lote = JSON.parse(p.lote);
    if (!Array.isArray(lote) || lote.length < 1) throw new Error('El lote GPS está vacío o no es válido.');
    if (lote.length > 20) throw new Error('El lote supera el máximo permitido de 20 puntos.');
    const filas = lote.map((item, indice) => {
      const lat = Number(String(item.latitud == null ? '' : item.latitud).replace(',', '.'));
      const lng = Number(String(item.longitud == null ? '' : item.longitud).replace(',', '.'));
      if (!Number.isFinite(lat) || !Number.isFinite(lng) || lat < -90 || lat > 90 || lng < -180 || lng > 180) {
        throw new Error('Coordenadas inválidas en el punto ' + (indice + 1));
      }
      const fecha = item.timestamp ? new Date(item.timestamp) : new Date();
      const precisionRaw = String(item.precision == null ? '' : item.precision).trim();
      const rumboRaw = String(item.rumbo == null ? '' : item.rumbo).trim();
      return [!isNaN(fecha.getTime()) ? fecha : new Date(),
        String(item.usuario || 'sin_usuario').slice(0, 150),
        String(item.nombre || item.usuario || 'Chofer').slice(0, 150), String(item.zona || '').slice(0, 150),
        lat, lng, precisionRaw !== '' && Number.isFinite(Number(precisionRaw)) ? Number(precisionRaw) : '',
        rumboRaw !== '' && Number.isFinite(Number(rumboRaw)) ? Number(rumboRaw) : '',
        String(item.activo == null ? '1' : item.activo).trim() === '0' ? 0 : 1];
    });
    lock.waitLock(20000);
    const hoja = dvHojaGPS_();
    const primeraFila = hoja.getLastRow() + 1;
    hoja.getRange(primeraFila, 1, filas.length, 9).setValues(filas);
    SpreadsheetApp.flush();
    return respuestaJSON({ok:true, accion:'registrarGPSLote', recibidos:filas.length,
      primeraFila:primeraFila, ultimaFila:primeraFila + filas.length - 1});
  } catch (error) {
    console.error('Error registrarGPSLote: ' + (error.stack || error.message));
    return respuestaJSON({ok:false, accion:'registrarGPSLote', error:error.message});
  } finally { try { lock.releaseLock(); } catch (error) {} }
}

function dvHojaGPS_() {
  const libro = SpreadsheetApp.openById(dvConfiguracion_('ID_HOJA_DE_DISPERSION'));
  let hoja = libro.getSheetByName('GPS Día Verde');
  if (!hoja) hoja = libro.insertSheet('GPS Día Verde');
  if (hoja.getLastRow() === 0) {
    hoja.appendRow(['Timestamp', 'Usuario', 'Nombre', 'Zona', 'Latitud', 'Longitud', 'Precision', 'Rumbo', 'Activo']);
  }
  return hoja;
}

function registrarGPS(e) {
  const lock = LockService.getScriptLock();
  try {
    const p = e && e.parameter ? e.parameter : {};
    const latitud = Number(String(p.latitud == null ? '' : p.latitud).replace(',', '.'));
    const longitud = Number(String(p.longitud == null ? '' : p.longitud).replace(',', '.'));
    if (!Number.isFinite(latitud) || !Number.isFinite(longitud) ||
        latitud < -90 || latitud > 90 || longitud < -180 || longitud > 180) {
      throw new Error('Latitud o longitud ausente o inválida.');
    }
    lock.waitLock(20000);
    const hoja = dvHojaGPS_();
    const fecha = p.timestamp ? new Date(p.timestamp) : new Date();
    const precisionRaw = String(p.precision == null ? '' : p.precision).trim();
    const rumboRaw = String(p.rumbo == null ? '' : p.rumbo).trim();
    const activo = String(p.activo == null ? '1' : p.activo).trim() === '0' ? 0 : 1;
    hoja.appendRow([!isNaN(fecha.getTime()) ? fecha : new Date(),
      String(p.usuario || 'sin_usuario').slice(0, 150), String(p.nombre || p.usuario || 'Chofer').slice(0, 150),
      String(p.zona || '').slice(0, 150), latitud, longitud,
      precisionRaw !== '' && Number.isFinite(Number(precisionRaw)) ? Number(precisionRaw) : '',
      rumboRaw !== '' && Number.isFinite(Number(rumboRaw)) ? Number(rumboRaw) : '', activo]);
    SpreadsheetApp.flush();
    return respuestaJSON({ok:true, accion:'registrarGPS', fila:hoja.getLastRow(), latitud:latitud,
      longitud:longitud, activo:activo, mensaje:'GPS guardado correctamente.'});
  } catch (error) {
    console.error('Error registrarGPS: ' + (error.stack || error.message));
    return respuestaJSON({ok:false, accion:'registrarGPS', error:error.message});
  } finally { try { lock.releaseLock(); } catch (error) {} }
}

function obtenerEvidenciasHoy() {
  try {
    const libro = SpreadsheetApp.openById(dvConfiguracion_('ID_HOJA_DE_DISPERSION'));
    const hoja = libro.getSheetByName(HOJA_HISTORICO);
    if (!hoja) return respuestaJSON({ok:false, error:"No existe la hoja '" + HOJA_HISTORICO + "'."});
    const valores = hoja.getDataRange().getValues();
    const hoy = obtenerFechaHoy();
    const evidencias = [];
    for (let i = 1; i < valores.length; i++) {
      const fila = valores[i];
      const fecha = convertirFecha(fila[0]);
      if (fecha !== hoy) continue;
      evidencias.push({fecha:fecha, hora:String(fila[1] || ''), zona:String(fila[2] || ''),
        direccion:String(fila[3] || ''), latitud:String(fila[4] || ''), longitud:String(fila[5] || ''),
        foto:String(fila[6] || ''), estado:String(fila[7] || ''), idEvidencia:String(fila[8] || '')});
    }
    return respuestaJSON({ok:true, fecha:hoy, evidencias:evidencias});
  } catch (error) { return respuestaJSON({ok:false, error:error.message}); }
}

function obtenerFechaHoy() {
  return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd/MM/yyyy');
}

function normalizarFecha(valor) {
  if (valor instanceof Date && !isNaN(valor.getTime())) {
    return Utilities.formatDate(valor, Session.getScriptTimeZone(), 'dd/MM/yyyy');
  }
  const texto = String(valor || '').trim();
  if (!texto) return '';
  let partes = texto.split('/');
  if (partes.length === 3 && /^\d{1,2}$/.test(partes[0]) && /^\d{1,2}$/.test(partes[1]) && /^\d{4}$/.test(partes[2])) {
    return partes[0].padStart(2, '0') + '/' + partes[1].padStart(2, '0') + '/' + partes[2];
  }
  partes = texto.split('-');
  if (partes.length === 3 && /^\d{1,2}$/.test(partes[0]) && /^\d{1,2}$/.test(partes[1]) && /^\d{4}$/.test(partes[2])) {
    return partes[0].padStart(2, '0') + '/' + partes[1].padStart(2, '0') + '/' + partes[2];
  }
  if (partes.length === 3 && /^\d{4}$/.test(partes[0]) && /^\d{1,2}$/.test(partes[1]) && /^\d{1,2}$/.test(partes[2])) {
    return partes[2].padStart(2, '0') + '/' + partes[1].padStart(2, '0') + '/' + partes[0];
  }
  const fechaTexto = texto.split(' ')[0];
  if (fechaTexto !== texto) return normalizarFecha(fechaTexto);
  return texto;
}

function convertirFecha(valor) { return normalizarFecha(valor); }

function normalizarClave(valor) {
  return String(valor || '').trim().toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '').replace(/\s+/g, ' ');
}

function obtenerOCrearCarpeta(carpetaPadre, nombre) {
  const nombreLimpio = String(nombre || 'Sin nombre').trim();
  const carpetas = carpetaPadre.getFoldersByName(nombreLimpio);
  return carpetas.hasNext() ? carpetas.next() : carpetaPadre.createFolder(nombreLimpio);
}

/** Modulo de ingreso incorporado en este archivo completo. */
function dvAuthJSON_(datos) {
  datos.authVersion = 2;
  return ContentService.createTextOutput(JSON.stringify(datos))
    .setMimeType(ContentService.MimeType.JSON);
}

function dvAuthHash_(texto, secreto) {
  return Utilities.base64EncodeWebSafe(
    Utilities.computeHmacSha256Signature(texto, secreto, Utilities.Charset.UTF_8)
  );
}

function dvAuthIguales_(a, b) {
  a = String(a); b = String(b);
  var diferencia = a.length ^ b.length;
  for (var i = 0; i < Math.max(a.length, b.length); i++) {
    diferencia |= (a.charCodeAt(i) || 0) ^ (b.charCodeAt(i) || 0);
  }
  return diferencia === 0;
}

/** Ejecutar desde el editor, nunca desde doGet/doPost.
 * Antes: cargar DV_AUTH_IMPORT en Propiedades de la secuencia de comandos.
 * Formato JSON: [{"usuario":"...","password":"...","nombre":"..."}]
 * Se conserva cada cuenta y clave; se elimina el JSON original tras importar.
 */
function importarUsuariosDiaVerde() {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var props = PropertiesService.getScriptProperties();
    var lista = JSON.parse(props.getProperty('DV_AUTH_IMPORT') || 'null');
    if (!Array.isArray(lista) || !lista.length) throw new Error('Falta DV_AUTH_IMPORT.');
    var secreto = props.getProperty('DV_AUTH_SECRET') ||
      Utilities.getUuid() + Utilities.getUuid() + Utilities.getUuid();
    var usuarios = {};
    lista.forEach(function(u) {
      var nombreUsuario = String(u.usuario || '').trim();
      var clave = nombreUsuario.toLowerCase();
      if (!clave || !u.password || Object.prototype.hasOwnProperty.call(usuarios, clave) ||
          ['__proto__', 'constructor', 'prototype'].indexOf(clave) >= 0) {
        throw new Error('La lista contiene cuentas inválidas o repetidas.');
      }
      var sal = Utilities.getUuid();
      usuarios[clave] = { usuario: nombreUsuario, nombre: String(u.nombre || nombreUsuario),
        sal: sal, hash: dvAuthHash_(sal + '\n' + String(u.password), secreto) };
    });
    var serializado = JSON.stringify(usuarios);
    if (Utilities.newBlob(serializado).getBytes().length > 8500) {
      throw new Error('La lista supera el tamaño admitido para esta configuración.');
    }
    props.setProperties({DV_AUTH_SECRET: secreto, DV_AUTH_USERS: serializado});
    props.deleteProperty('DV_AUTH_IMPORT');
    console.log('Importación terminada: ' + lista.length + ' cuentas.');
  } finally { lock.releaseLock(); }
}

/** Devuelve null para las operaciones existentes de evidencias. */
function dvProcesarAuth_(e) {
  var datos;
  try { datos = JSON.parse(e && e.postData && e.postData.contents || '{}'); }
  catch (error) { return null; }
  if (['diaVerdeLogin', 'diaVerdeSesion', 'diaVerdeLogout'].indexOf(datos.accion) < 0) return null;
  try {
    var cache = CacheService.getScriptCache();
    var token = String(datos.token || '');
    if (datos.accion !== 'diaVerdeLogin') {
      if (!/^[a-f0-9-]{72}$/.test(token)) return dvAuthJSON_({ok:false});
      var sesionKey = 'dvSession:' + token;
      if (datos.accion === 'diaVerdeLogout') {
        cache.remove(sesionKey);
        return dvAuthJSON_({ok:true});
      }
      var sesion = cache.get(sesionKey);
      return dvAuthJSON_(sesion ? {ok:true, usuario:JSON.parse(sesion)} : {ok:false});
    }
    var usuario = String(datos.usuario || '').trim().toLowerCase();
    var password = String(datos.password || '');
    if (!usuario || usuario.length > 150 || !password || password.length > 512) {
      return dvAuthJSON_({ok:false, error:'Usuario o contraseña incorrectos.'});
    }
    var lock = LockService.getScriptLock();
    lock.waitLock(10000);
    try {
      var props = PropertiesService.getScriptProperties();
      var secreto = props.getProperty('DV_AUTH_SECRET');
      var usuariosJSON = props.getProperty('DV_AUTH_USERS');
      if (!secreto || !usuariosJSON) {
        return dvAuthJSON_({ok:false, error:'El ingreso todavía no fue configurado en el servidor.'});
      }
      var intentosKey = 'dvAttempts:' + dvAuthHash_(usuario, secreto);
      var registro = JSON.parse(cache.get(intentosKey) || '{"cantidad":0,"hasta":0}');
      if (registro.hasta > Date.now()) {
        return dvAuthJSON_({ok:false, error:'Demasiados intentos. Esperá 15 minutos y volvé a intentar.'});
      }
      var usuarios = JSON.parse(usuariosJSON);
      var u = Object.prototype.hasOwnProperty.call(usuarios, usuario) ? usuarios[usuario] : null;
      var hash = dvAuthHash_((u ? u.sal : 'cuenta-ausente') + '\n' + password, secreto);
      if (!u || !dvAuthIguales_(u.hash, hash)) {
        registro.cantidad++;
        if (registro.cantidad >= 10) registro.hasta = Date.now() + 900000;
        cache.put(intentosKey, JSON.stringify(registro), 900);
        return dvAuthJSON_({ok:false, error:'Usuario o contraseña incorrectos.'});
      }
      cache.remove(intentosKey);
      var perfil = {usuario:u.usuario, nombre:u.nombre};
      token = Utilities.getUuid() + Utilities.getUuid();
      cache.put('dvSession:' + token, JSON.stringify(perfil), 21600);
      return dvAuthJSON_({ok:true, token:token, usuario:perfil});
    } finally { lock.releaseLock(); }
  } catch (error) {
    return dvAuthJSON_({ok:false, error:'No se pudo validar el ingreso. Volvé a intentar.'});
  }
}
